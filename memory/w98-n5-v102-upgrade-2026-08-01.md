-- 件 7 真定义: SearchLog 表回收率 = clicks / total_searches ≥ 30%

       count(*) FILTER (WHERE clicked) * 100.0 / NULLIF(count(*), 0) AS ctr
FROM search_logs GROUP BY 1 ORDER BY 1 DESC LIMIT 14;
